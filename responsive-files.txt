type Props = { params: Promise<{ slug: string }> }

export async function generateMetadata({ params }: Props) {
  const { slug } = await params
  const project = await getProject(slug)
  if (!project) return { title: 'Not found', robots: { index: false } }
  const where = project.location ? ` in ${project.location}` : ''
  const description =
    project.description ||
    `${project.title}: a ${project.category.toLowerCase()} photography story${where} by MAVX Frames, Chennai.`
  return {
    title: `${project.title} — ${project.category} Photography${where}`,
    description,
    alternates: { canonical: `/work/${project.slug}` },
    openGraph: { type: 'article', title: project.title, description, images: project.cover ? [{ url: project.cover, alt: project.title }] : undefined },
    twitter: { card: 'summary_large_image', title: project.title, description, images: project.cover ? [project.cover] : undefined }
  }
}

export default async function ProjectPage({ params }: Props) {
  const { slug } = await params
  const projects = await getProjects()
  const index = projects.findIndex((p) => p.slug === slug)
  if (index === -1) notFound()
  const project = projects[index]
  const next = projects.length > 1 ? projects[(index + 1) % projects.length] : null

  return (
    <main>
      <JsonLd
        data={[
          {
            '@context': 'https://schema.org',
            '@type': 'ImageGallery',
            name: project.title,
            description: project.description || undefined,
            url: `${SITE_URL}/work/${project.slug}`,
            contentLocation: project.location || undefined,
            image: [project.cover, ...(project.photos || [])].filter(Boolean).slice(0, 12),

              { '@type': 'ListItem', position: 2, name: 'Work', item: `${SITE_URL}/work` },
              { '@type': 'ListItem', position: 3, name: project.title, item: `${SITE_URL}/work/${project.slug}` }
            ]
          }
        ]}

      <ProjectDetail project={project} next={next} />
    </main>
  )
}

import { useEffect, useRef, useState } from 'react'
import gsap from 'gsap'
import { MoonIcon, SunIcon } from './Icons'

export default function ThemeToggle() {
  const [dark, setDark] = useState(false)
  const icon = useRef<HTMLSpanElement>(null)

  useEffect(() => {
    setDark(document.documentElement.classList.contains('dark'))
    const media = window.matchMedia('(prefers-color-scheme: dark)')
    const sync = () => {
      let saved: string | null = null
      try { saved = localStorage.getItem('theme') } catch {}
      const next = saved ? saved === 'dark' : media.matches
      document.documentElement.classList.toggle('dark', next)
      setDark(next)
    }
    media.addEventListener('change', sync)
    window.addEventListener('storage', sync)
    return () => { media.removeEventListener('change', sync); window.removeEventListener('storage', sync) }
  }, [])

  const toggle = () => {
    const next = !dark
    document.documentElement.classList.toggle('dark', next)
    try {
      localStorage.setItem('theme', next ? 'dark' : 'light')
    } catch {}
    setDark(next)
    if (icon.current && !window.matchMedia('(prefers-reduced-motion: reduce)').matches) {
      gsap.fromTo(
        icon.current,
        { rotate: -70, scale: 0.6, opacity: 0 },
        { rotate: 0, scale: 1, opacity: 1, duration: 0.6, ease: 'power3.out', clearProps: 'all' }
      )
    }
  }

  return (
    <button
      onClick={toggle}
      aria-label={dark ? 'Switch to light mode' : 'Switch to dark mode'}
      title={dark ? 'Light mode' : 'Dark mode'}
      className="nav-icon-btn"
    >
      <span ref={icon} className="flex">
        {dark ? <SunIcon width={17} height={17} /> : <MoonIcon width={17} height={17} />}
      </span>
    </button>
  )
}

export default function ContactModal({
  open,
  onClose

  open: boolean
  onClose: () => void
}) {
  const panel = useRef<HTMLDivElement>(null)
  const [form, setForm] = useState({ name: '', contact: '', type: '', message: '' })
  const [status, setStatus] = useState<Status>('idle')
  const [error, setError] = useState('')

  useEffect(() => {
    if (!open) return
    const onKey = (e: KeyboardEvent) => e.key === 'Escape' && onClose()
    window.addEventListener('keydown', onKey)
    return () => window.removeEventListener('keydown', onKey)
  }, [open, onClose])

  useGSAP(
    () => {
      if (open && panel.current) {
        gsap.fromTo(
          panel.current,
          { y: 40, autoAlpha: 0, scale: 0.98 },
          { y: 0, autoAlpha: 1, scale: 1, duration: 0.6, ease: 'power3.out' }
        )

'use client'

import { createContext, useCallback, useContext, useMemo, useState } from 'react'
import { usePathname, useRouter, useSearchParams } from 'next/navigation'
import ContactModal from './ContactModal'
import ProjectModal from './ProjectModal'
import type { Project } from '@/lib/types'

interface UI {
  isEdit: boolean
  href: (path: string, hash?: string) => string
  openContact: () => void
  openEditor: (project: Project | null) => void
}

const Ctx = createContext<UI | null>(null)

export function useUI() {
  const ctx = useContext(Ctx)
  if (!ctx) throw new Error('useUI must be used inside UIProvider')
  return ctx
}

export default function UIProvider({ children }: { children: React.ReactNode }) {
  const params = useSearchParams()
  const router = useRouter()
  const pathname = usePathname()
  const isEdit = params.get('edit') === 'true'

  const [contactOpen, setContactOpen] = useState(false)
  const [editor, setEditor] = useState<{ open: boolean; project: Project | null }>({
    open: false,
    project: null
  })

  const href = useCallback(
    (path: string, hash?: string) =>
      `${path}${isEdit ? '?edit=true' : ''}${hash ? `#${hash}` : ''}`,
    [isEdit]
  )

  const openContact = useCallback(() => setContactOpen(true), [])
  const openEditor = useCallback(
    (project: Project | null) => setEditor({ open: true, project }),
    []
  )

  const value = useMemo(
    () => ({ isEdit, href, openContact, openEditor }),
    [isEdit, href, openContact, openEditor]
  )

  return (
    <Ctx.Provider value={value}>
      {children}
      <ContactModal open={contactOpen} onClose={() => setContactOpen(false)} />
      <ProjectModal
        open={editor.open}
        project={editor.project}
        onClose={() => setEditor((s) => ({ ...s, open: false }))}
        onSaved={() => {
          setEditor((s) => ({ ...s, open: false }))
          router.refresh()
        }}
        onDeleted={() => {
          setEditor((s) => ({ ...s, open: false }))
          if (pathname.startsWith('/work/')) router.push(href('/work'))
          router.refresh()
        }}
      />
    </Ctx.Provider>
  )
}
